# Section B # Q7: Detailed EDA import pandas as pd import numpy as np import matplotlib.pyplot as plt import seaborn as sns from sklearn.preprocessing import OneHotEncoder, StandardScaler, MinMaxScaler # Load dataset # Reading the dataset into a pandas DataFrame data = pd.read_csv('train_Feb2025.csv') # Display basic information about the dataset (column names, non-null values, data types) display(data.info()) display(data.describe()) # Check for missing values in the dataset missing_values = data.isnull().sum() missing_values = missing_values[missing_values > 0] display(missing_values) # Display only columns with missing values # Visualizing the target variable 'SalePrice' plt.figure(figsize=(10, 5)) sns.histplot(data['SalePrice'], kde=True, bins=30) plt.title("Distribution of SalePrice") plt.xlabel("Sale Price") plt.ylabel("Frequency") plt.show() # Correlation heatmap to understand relationships between numerical variables plt.figure(figsize=(12, 8)) numeric_data = data.select_dtypes(include=[np.number]) # Select only numerical columns sns.heatmap(numeric_data.corr(), cmap='coolwarm', annot=False) plt.title("Correlation Heatmap") plt.show() ########################################################################################### # Q8: Dummy Coding for Categorical Variables # Identifying categorical columns for encoding categorical_cols = data.select_dtypes(include=['object']).columns # Applying One-Hot Encoding to categorical variables dummy_data = pd.get_dummies(data, columns=categorical_cols, drop_first=True) display(dummy_data.head()) # Displaying the first few rows of the transformed dataset ########################################################################################### # Q9: Outlier Detection and Treatment # Using the IQR method to identify outliers # Selecting only numerical columns for outlier detection numeric_data = data.select_dtypes(include=[np.number]) Q1 = numeric_data.quantile(0.25) # First quartile Q3 = numeric_data.quantile(0.75) # Third quartile IQR = Q3 - Q1 # Interquartile range # Identifying outliers as values that fall outside 1.5*IQR range outlier_mask = (numeric_data < (Q1 - 1.5 * IQR)) | (numeric_data > (Q3 + 1.5 * IQR)) # Marking outliers as NaN for handling later, but only in the numerical columns data[outlier_mask] = np.nan display(data.isnull().sum()) # Checking missing values after marking outliers ########################################################################################### # Q10: Missing Values Treatment # Filling missing values based on data type for col in data.columns: if data[col].dtype == 'object': data[col].fillna(data[col].mode()[0], inplace=True) # Filling categorical variables with mode (most frequent value) else: data[col].fillna(data[col].median(), inplace=True) # Filling numerical variables with median display(data.isnull().sum()) # Ensuring no missing values remain ########################################################################################### # Q11: Scaling or Normalization # Standardizing numerical features using StandardScaler scaler = StandardScaler() numeric_cols = data.select_dtypes(include=[np.number]).columns # Selecting numerical columns scaled_data = scaler.fit_transform(data[numeric_cols]) data[numeric_cols] = scaled_data # Updating dataset with scaled values display(data.head()) # Display first few rows after scaling # Save processed dataset to a new CSV file data.to_csv("processed_train_Feb2025.csv", index=False) ###########################################################################################
